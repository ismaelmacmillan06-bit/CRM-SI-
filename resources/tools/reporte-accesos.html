Reporte de Accesos — Generador

Uso y conexión a MEE por colegio

  1. Entra a Users dentro de MEE.
  2. Clic derecho → Inspeccionar → pestaña Console.
  3. Pega el script de abajo y presiona Enter.
  4. Espera a que termine de recorrer a todos los usuarios (verás el avance en la consola) — al terminar descarga un CSV automáticamente.
  5. Sube ese CSV en la columna de la derecha.
(async () => {
  // ====== CONFIGURACIÓN — edita esto según tu panel ======
  const CONFIG = {
    listUrlTemplate: (page, limit) => `${location.origin}/users?page=${page}&limit=${limit}`,
    pageLimit: 100,
    detailLinkSelector: '[data-testid="user-view-link"]',
    detailLinkText: null,
    fields: {
      usuario: '[data-testid="user-username"]',
      ultima_conexion: '[data-testid="user-last-login"]',
      rol: '[data-testid="role-content"] .role-title',
      clases: { selector: '[data-testid="classes-table"] .class-name', joinWith: '; ' },
    },
    paginationRegex: /\d+-\d+\s+(?:de|of)\s+(\d+)/i,
    concurrency: 8,
    outputFileName: 'export.csv',
  };
  // ========================================================

  async function fetchDoc(url) {
    const res = await fetch(url, { credentials: 'include' });
    const html = await res.text();
    return new DOMParser().parseFromString(html, 'text/html');
  }

  function extractField(doc, def) {
    if (typeof def === 'string') {
      return doc.querySelector(def)?.textContent.trim() ?? '';
    }
    const els = [...doc.querySelectorAll(def.selector)];
    return els.map(el => el.textContent.trim()).filter(Boolean).join(def.joinWith ?? '; ');
  }

  // 1. Detectar total de páginas
  let doc = await fetchDoc(CONFIG.listUrlTemplate(1, CONFIG.pageLimit));
  let total = null;
  const paginationText = [...doc.querySelectorAll('*')]
    .map(el => el.textContent)
    .find(t => CONFIG.paginationRegex.test(t || ''));
  if (paginationText) total = parseInt(paginationText.match(CONFIG.paginationRegex)[1], 10);
  const totalPages = total ? Math.ceil(total / CONFIG.pageLimit) : 1;
  console.log(`Detectados ${total ?? '??'} registros, ${totalPages} páginas`);

  // 2. Recolectar links de detalle en todas las páginas
  let links = [];
  for (let page = 1; page <= totalPages; page++) {
    const pageDoc = page === 1 ? doc : await fetchDoc(CONFIG.listUrlTemplate(page, CONFIG.pageLimit));
    const pageLinks = [...pageDoc.querySelectorAll(CONFIG.detailLinkSelector)]
      .filter(a => !CONFIG.detailLinkText || a.textContent.trim() === CONFIG.detailLinkText)
      .map(a => new URL(a.getAttribute('href'), location.origin).href);
    links.push(...pageLinks);
    console.log(`Página ${page}: ${pageLinks.length} registros`);
  }
  console.log(`Total recolectado: ${links.length}`);

  // 3. Visitar cada detalle con concurrencia limitada
  const results = [];
  let idx = 0;
  async function worker() {
    while (idx < links.length) {
      const myIdx = idx++;
      const url = links[myIdx];
      try {
        const udoc = await fetchDoc(url);
        const row = {};
        for (const [key, def] of Object.entries(CONFIG.fields)) {
          row[key] = extractField(udoc, def);
        }
        results.push(row);
      } catch (e) {
        console.error('Error en', url, e);
      }
      if (myIdx % 25 === 0) console.log(`Procesados ${myIdx}/${links.length}`);
    }
  }
  await Promise.all(Array.from({ length: CONFIG.concurrency }, worker));

  // 4. Generar CSV y descargar
  const headers = Object.keys(CONFIG.fields);
  const csvRows = [
    headers.join(','),
    ...results.map(r => headers.map(h => `"${(r[h] || '').replace(/"/g, '""')}"`).join(',')),
  ];
  const blob = new Blob(['﻿' + csvRows.join('\n')], { type: 'text/csv;charset=utf-8;' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = CONFIG.outputFileName;
  a.click();

  console.log(`Listo: ${results.length} registros exportados a ${CONFIG.outputFileName}.`);
})();

Completa los datos del colegio y carga el archivo exportado de MEE para generar el reporte.

Sin logo seleccionado PNG o JPG · se usará en el encabezado del reporte

Nuestro equipo entra a todos los perfiles para validar el acceso antes de entregar el colegio. Indica siempre esta fecha — si la dejas vacía, esas conexiones de verificación se contarán como uso real y el reporte dirá que todos ya entraron aunque no sea cierto.

Ningún archivo seleccionado Columnas esperadas: usuario, ultima_conexion, rol, clases (.csv, .xlsx o .xls)

Es el CSV que descarga el script de la consola — súbelo tal cual, sin modificarlo.

Todo el procesamiento ocurre en este navegador: el archivo y los datos de alumnos no se suben a ningún servidor.
